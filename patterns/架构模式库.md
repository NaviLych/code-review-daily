```javascript
function UserList({ users }) {

    const [sortField, setSortField] = useState('name');

            .sort((a, b) => sortField === 'name'

        <div>
            <SearchInput value={searchTerm} onChange={setSearchTerm} />
            <SortSelect value={sortField} onChange={setSortField} />
            <UserCardList users={processedUsers} />
        </div>
    );
}
```

**为什么好**：

```javascript
// ✅ 好设计：完整的副作用管理
function ChatRoom({ roomId }) {
    const [messages, setMessages] = useState([]);
    const wsRef = useRef(null);
    
    useEffect(() => {
        let isMounted = true;
        let reconnectTimer = null;
        
        const connect = () => {
            const ws = new WebSocket(`wss://api.example.com/chat/${roomId}`);
            wsRef.current = ws;
            
            ws.onmessage = (event) => {
                if (isMounted) {
                    setMessages(prev => [...prev, JSON.parse(event.data)]);
                }
            };
            
            ws.onclose = () => {
                if (isMounted) {
                    reconnectTimer = setTimeout(connect, 3000);
                }
            };
        };
        
        connect();
        
        // 清理函数
        return () => {
            isMounted = false;
            clearTimeout(reconnectTimer);
            if (wsRef.current?.readyState === WebSocket.OPEN) {
                wsRef.current.close();
            }
        };
    }, [roomId]);
    
    return <MessageList messages={messages} />;
}
```

**为什么好**：
- isMounted标志防止卸载后更新
- 重连机制提升可靠性
- 清理函数完整，无内存泄漏
- roomId变化时正确重建连接

#### 2.2 防御性SQL查询

```python
# ✅ 好设计：参数化查询，防止注入
import sqlite3
from typing import Optional

class UserRepository:
    def __init__(self, db_path: str):
        self.db_path = db_path
    
    def find_by_email(self, email: str) -> Optional[dict]:
        """查找用户，参数化查询防止SQL注入"""
        # ✅ 参数化查询
        query = "SELECT id, email, created_at FROM users WHERE email = ?"
        
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.cursor()
            cursor.execute(query, (email,))  # ✅ 参数单独传递
            row = cursor.fetchone()
            
            if row:
                return {
                    'id': row[0],
                    'email': row[1],
                    'created_at': row[2]
                }
        return None
    
    def create_user(self, email: str, password_hash: str) -> int:
        """创建用户，输入验证"""
        # ✅ 输入验证
        if not email or '@' not in email:
            raise ValueError("Invalid email format")
        
        if len(password_hash) < 60:  # bcrypt hash长度
            raise ValueError("Invalid password hash")
        
        query = """
            INSERT INTO users (email, password_hash, created_at)
            VALUES (?, ?, datetime('now'))
        """
        
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.cursor()
            cursor.execute(query, (email, password_hash))
            return cursor.lastrowid
```

**为什么好**：
- 参数化查询防注入
- 输入验证显式
- 类型注解清晰
- 错误消息有意义

```python
# controllers/user_controller.py
from fastapi import Depends
from schemas.user_schema import UserCreate, UserResponse
from services.user_service import UserService

class UserController:
    def __init__(self, service: UserService = Depends()):
        self.service = service

from repositories.user_repository import UserRepository
from models.user import User

class UserService:
    def __init__(self, repository: UserRepository):
        self.repository = repository
    
    async def create_user(self, data) -> User:
        # 业务逻辑
        user = User.create(email=data.email)

class UserRepository:
    async def save(self, user: User) -> User:
        # 数据访问
        db_user = self.db.query(User).filter_by(email=user.email).first()
        if db_user:
            raise DuplicateEmailError()
        self.db.add(user)
        self.db.commit()
        return user
```

#### 3.2 错误边界与降级

```python
# ✅ 好设计：防御性编程与降级
from typing import TypeVar, Callable
from functools import wraps
import logging

logger = logging.getLogger(__name__)

T = TypeVar('T')

def with_fallback(fallback_value: T, exceptions: tuple = (Exception,)):
    """
    装饰器：异常时返回降级值
    """
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except exceptions as e:
                logger.error(f"{func.__name__} failed: {e}")
                return fallback_value
        return wrapper
    return decorator

# 使用示例
class CacheService:
    @with_fallback(fallback_value={})  # 缓存失败返回空字典
    def get(self, key: str) -> dict:
        return self.redis.get(key)
    
    @with_fallback(fallback_value=False)  # 写入失败返回False
    def set(self, key: str, value: dict, ttl: int) -> bool:
        return self.redis.setex(key, ttl, json.dumps(value))
```

**为什么好**：
- 异常处理显式
- 降级策略明确
- 日志记录问题
- 便于追踪

```python
# ✅ 好设计：上下文管理器
import threading
from contextlib import contextmanager

class DatabasePool:
    def __init__(self, max_connections: int = 10):
        self.max_connections = max_connections
        self._pool = []
        self._lock = threading.Lock()
        self._semaphore = threading.Semaphore(max_connections)
    
    @contextmanager
    def get_connection(self):
        """线程安全的连接获取"""
        self._semaphore.acquire()
        conn = None
        
        try:
            with self._lock:
                if self._pool:
                    conn = self._pool.pop()
                else:
                    conn = self._create_connection()
            
            yield conn
            
        finally:
            with self._lock:
                self._pool.append(conn)
            self._semaphore.release()
    
    def _create_connection(self):
        # 创建新连接
        pass

# 使用
with pool.get_connection() as conn:
    conn.execute("SELECT * FROM users")
```

**为什么好**：
- 线程安全
- 资源自动释放
- 避免连接泄漏

```python
# ✅ 好设计：支持取消的长任务
import asyncio
from typing import Optional

class LongRunningTask:
    def __init__(self):
        self._cancelled = False
    
    async def process(self, items: list, progress_callback=None):
        results = []
        
        for i, item in enumerate(items):
            # 检查取消标志
            if self._cancelled:
                logger.info(f"Task cancelled at item {i}")
                break
            
            result = await self._process_item(item)
            results.append(result)
            
            # 报告进度
            if progress_callback:
                progress_callback(i + 1, len(items))
        
        return results
    
    def cancel(self):
        self._cancelled = True

# 使用
task = LongRunningTask()

async def run_with_timeout():
    try:
        results = await asyncio.wait_for(
            task.process(large_dataset),
            timeout=60.0  # 60秒超时
        )
    except asyncio.TimeoutError:
        task.cancel()
        logger.warning("Task timed out")
```

```java
// ✅ 好设计：支付确认幂等性
public void confirm(String paymentKey, String method) {
    if (this.status == PaymentStatus.DONE) {
        return;  // 幂等：已处理的不重复处理
    }
    if (this.status != PaymentStatus.READY) {
        throw new IllegalStateException("只能在READY状态确认支付");
    }
    this.paymentKey = paymentKey;
    this.method = method;
    this.status = PaymentStatus.DONE;
}
```

**为什么好**：
- 双重检查：DONE状态幂等返回，非READY状态抛异常
- 状态机守卫：只允许READY→DONE转换
- 领域逻辑内聚：幂等性在Entity内部保障，不依赖Service层

```go
        return err
    })

    if err := g.Wait(); err != nil {
        return nil, fmt.Errorf("fetch dashboard for %s: %w", userID, err)
    }

    return &Dashboard{
        Profile:       profile,
        Orders:        orders,
        Notifications: notifications,
    }, nil
}

// ✅ 带并发限制的版本
func enrichProducts(ctx context.Context, products []Product) ([]EnrichedProduct, error) {
    results := make([]EnrichedProduct, len(products))
    g, gctx := errgroup.WithContext(ctx)
    g.SetLimit(20)  // ✅ 限制并发数

    for i, p := range products {
        i, p := i, p
        g.Go(func() error {
            enriched, err := enrichProduct(gctx, p)
            if err != nil {
                return fmt.Errorf("enrich product %s: %w", p.ID, err)
            }
            results[i] = enriched
            return nil
        })
    }

    return results, g.Wait()
}
```

**为什么好**：
- "all succeed or all cancel" 语义简洁有力
- context 取消自动级联，无需手动传播
- SetLimit 内置限流，无需额外 semaphore
- error 包装保留了上下文信息

```swift
            throw URLError(.timedOut)
        }
        let result = try await group.next()!
        group.cancelAll()
        return result
    }
```

### 坏设计 ❌

#### 8.1 事务范围膨胀（Transaction Scope Bloat）

```java
// ❌ 事务内混合了远程调用、审计、通知
@Transactional
public TransferResult transfer(Long fromId, Long toId, BigDecimal amount) throws IOException {
    Account from = accountRepository.findById(fromId).orElseThrow();
    from.debit(amount);
    accountRepository.save(from);
    complianceClient.checkAntiMoneyLaundering(from, amount);  // 远程调用！长事务
```

### 好设计 ✅

#### 8.2 事务边界分离：校验前置 + DB操作事务化 + 副作用后置

```java
// ✅ 三段式架构：校验→事务→事件
@Service
public class TransferService {
    @Autowired private AccountRepository accountRepository;
    @Autowired private ComplianceClient complianceClient;
    @Autowired private ApplicationEventPublisher eventPublisher;

    // 第一段：事务外校验（不占连接、不持有锁）
    public TransferResult transfer(Long fromId, Long toId, BigDecimal amount) throws IOException {
        Account from = accountRepository.findById(fromId)
                .orElseThrow(() -> new BusinessException("转出账户不存在"));
        complianceClient.checkAntiMoneyLaundering(from, amount);  // 事务外远程调用
        return doTransfer(fromId, toId, amount);  // 校验通过再开事务
    }

    // 第二段：事务内纯DB操作（最短持有时间）
    @Transactional(rollbackFor = Exception.class)
    public TransferResult doTransfer(Long fromId, Long toId, BigDecimal amount) {
        // 按ID排序加锁，防止死锁
        Long first = Math.min(fromId, toId), second = Math.max(fromId, toId);
        Account a1 = accountRepository.findByIdForUpdate(first);
        Account a2 = accountRepository.findByIdForUpdate(second);
        Account from = first.equals(fromId) ? a1 : a2;
        Account to = first.equals(toId) ? a1 : a2;
        from.debit(amount);
        to.credit(amount);
        eventPublisher.publishEvent(new TransferCompletedEvent(fromId, toId, amount));
        return TransferResult.success();
    }
}

// 第三段：事务提交后异步处理副作用
@Component
public class TransferEventHandler {
    @Autowired private AuditLogRepository auditRepository;
    @Autowired private NotificationService notificationService;

    @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
    @Transactional(propagation = Propagation.REQUIRES_NEW)  // 安全：外层事务已释放连接
    public void handleTransferCompleted(TransferCompletedEvent event) {
        auditRepository.save(new AuditLog(event));
        notificationService.notifyAsync(event.getToId(), "收到转账");
    }
}
```

**为什么好**：
- **事务最短化**：事务内只有2次SELECT FOR UPDATE + 2次UPDATE，毫秒级完成
- **远程调用前置**：校验失败不浪费事务资源，不占连接
- **副作用后置**：AFTER_COMMIT保证REQUIRES_NEW不与主事务争连接
- **死锁防护**：按ID排序加锁，A→B和B→A不会互相等待
- **回滚精确**：只有DB操作失败才回滚，校验失败直接返回

**架构原则**：
```
事务内只做不可补偿的操作（DB写入）
事务外做可重试的操作（远程调用）
事务后做可丢失的操作（审计、通知）
```

## 十、Java并发锁架构模式

### 10.1 锁选型决策树 ✅

```
需要线程安全？
├── 只需单操作原子性？
│   ├── 读多写少 → CopyOnWriteArrayList / ConcurrentHashMap
│   └── 队列场景 → ConcurrentLinkedQueue / LinkedBlockingQueue
├── 需要复合操作原子性？
│   ├── 简单check-then-act → ConcurrentHashMap.computeIfAbsent
│   └── 复杂事务 → ReentrantLock + tryLock(timeout)
└── 需要跨资源协调？
    ├── 同一JVM → ReentrantLock + 锁排序（防死锁）
    └── 跨服务 → 分布式锁（Redis/Zookeeper）+ 超时
```

### 10.2 Virtual Thread锁兼容性 ✅

| 锁类型 | Virtual Thread兼容 | 原因 |
|--------|-------------------|------|
| synchronized | ❌ 不兼容 | pin载体线程，阻塞ForkJoinPool |
| ReentrantLock | ✅ 兼容 | 不pin，VT正常调度卸载 |
| java.util.concurrent锁 | ✅ 兼容 | LockSupport.park感知VT |
| Object.wait/notify | ❌ 不兼容 | 基于synchronized |
| Condition.await | ✅ 兼容 | 基于LockSupport |

### 10.3 锁粒度分层 ✅

| 级别 | 粒度 | 适用场景 | 风险 |
|------|------|----------|------|
| L1 | 方法级synchronized | 极简单场景 | Virtual Thread pinning |
| L2 | 实例级ReentrantLock | 单资源保护 | 锁内IO阻塞 |
| L3 | 资源ID级ConcurrentHashMap<Lock> | 多资源并发 | 锁泄漏/未释放 |
| L4 | 行级DB锁(SELECT FOR UPDATE) | 数据库事务 | 死锁检测 |
| L5 | 无锁(CAS/Atomic*) | 计数器/状态标记 | ABA问题 |

| 操作 | 释放锁 | 响应中断 | 适用场景 |
|------|--------|----------|----------|

```text
  -> started_at / finished_at

risk_batch_failure
  -> batch_id
  -> item_id
  -> error_type
  -> error_message
  -> retry_count

worker
  -> load page by cursor
  -> process with bounded executor
  -> write result idempotently
  -> save progress
```

**为什么好**：
- 能回答“批次停在哪、坏在哪、如何重来”
- 分页游标避免一次性加载全量数据
- 失败明细支持精准补偿
- 幂等结果表避免重复重算造成二次污染

**适用场景**：
- 风控重算
- 历史交易补偿
- 数据迁移
- 对账修复

### 10.7 线程池隔离模式 ✅

```java
@Bean
ThreadPoolTaskExecutor riskBatchExecutor() {
    ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
    executor.setThreadNamePrefix("risk-batch-");
    executor.setCorePoolSize(8);
    executor.setMaxPoolSize(16);
    executor.setQueueCapacity(500);
    executor.initialize();
    return executor;
}
```

**为什么好**：
- 批处理任务不挤占 Web 请求线程
- 线程名前缀让日志和监控可定位
- core/max/queue 显式表达资源预算
- 与下游 QPS、数据库连接池容量可以一起调参

**反面教材**：
```java
CompletableFuture.runAsync(() -> blockingHttpCall());
```

默认线程池适合 CPU 型小任务，不适合作为银行后台阻塞 IO 批处理的资源入口。

## 十一、设计原则速查

| 原则 | 含义 | 好设计 | 坏设计 |
|------|------|--------|--------|
| **SRP** | 单一职责 | UserController只处理请求 | DataManager处理一切 |
| **OCP** | 开闭原则 | 新功能通过扩展实现 | 修改现有类 |
| **LSP** | 里氏替换 | 子类可替换父类 | 子类破坏父类行为 |
| **ISP** | 接口隔离 | 多个小接口 | 一个大接口 |
| **DIP** | 依赖倒置 | 依赖抽象 | 依赖具体实现 |
| **KISS** | 简单原则 | 简单解决方案 | 过度工程化 |
| **YAGNI** | 不需要原则 | 按需实现 | 提前实现可能需要的功能 |
| **DRY** | 不重复 | 共享逻辑抽离 | 重复代码 |
