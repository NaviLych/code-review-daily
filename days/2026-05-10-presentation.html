2.4 🟡 Swift: 串行 await vs 并行 async let 性能陷阱

let payment = await getPayment(of: user)

// ✅ 并行：总耗时 = max(t1, t2, t3)

2.5 🔴 Swift: Task 隐式捕获 self 内存泄漏 内存泄漏

// ❌ Task 持有 ViewModel 引用 + 无限序列 → 永不释放
@MainActor
class ViewModel {
    func listenToNotifications() {
        Task {
            let notifications = NotificationCenter.default.notifications(
                named: UIDevice.orientationDidChangeNotification
            )
            for await notification in notifications {
                self.handle(notification)  // ⚠️ 强引用 self
            }
        }
    }
}

// ✅ 修复：[weak self] 在循环内部 unwrap
Task { [weak self] in
    guard let self else { return }
    for await notification in notifications {
        self.handle(notification)  // ✅ 可以正常结束
    }
}

}

// ✅ 优先使用普通 Task
Task { [weak self] in /* 直接访问 @MainActor 属性 */ }

三、Java 25 新特性陷阱

3.1 🔴 ScopedValue 不传播到线程池 新特性陷阱

来源: Your AI can write Java 25 right with SonarQube

// ❌ ScopedValue 绑定不传播到 ExecutorService 线程
ScopedValue.where(REQUEST_ID, id).run(() -> {
    executor.submit(() -> {  // ⚠️ 线程来自线程池
        String reqId = REQUEST_ID.get(); // NoSuchElementException!
        processAsync(reqId);
    });
});

// ✅ 使用 StructuredTaskScope 创建子任务
ScopedValue.where(REQUEST_ID, id).call(() -> {
    try (var scope = StructuredTaskScope.open(Joiner.allSuccessfulOrThrow())) {
        scope.fork(() -> {
            String reqId = REQUEST_ID.get(); // ✅ 正确继承
            processAsync(reqId); return null;
        });
        scope.join();
    }
    return null;
});

// ✅ 必须消费 Carrier

6.1 🔴 Double-Checked Locking 无 volatile 经典陷阱

来源: SonarSource - GPT-5.5's biggest blind spot

// ❌ 可能返回部分构造对象
public static Resource getInstance() {
    if (resource == null) {
        synchronized (ResourceFactory.class) {
            if (resource == null)
                resource = new Resource();  // ⚠️ JVM 重排序
        }
    }

// ✅ 方案1: synchronized 方法 (推荐)

    public static Resource resource = new Resource();

6.2 🔴 Boolean/FInteger 上 synchronize 隐蔽陷阱

// ✅ 专用锁对象
private final Object lock = new Object();
synchronized (lock) { /* 安全 */ }
扩展：Integer、String、List.of() 同样有缓存/不可变陷阱，切勿用作锁对象

✅ Swift Task 内存泄漏修复 & Task.detached 滥用

Task 内存泄漏修复

@MainActor
class ViewModel {
    func listenToNotifications() {
        Task { [weak self] in
            guard let self else { return }

            let notifications = NotificationCenter
                .default.notifications(
                named: UIDevice.orientationDidChangeNotification
            )

            for await notification in notifications {
                self.handle(notification)
                // ✅ 循环内部 unwrap
            }
        }
    }
}
💡 关键: [weak self] 必须在 for 循环内部 unwrap
