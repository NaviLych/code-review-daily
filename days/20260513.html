🔴 Go 错误变量遮蔽

严重度：HIGH · 来源：PVS-Studio

resp, err := c.List(opts)
if err != nil { return resp, err }
data := resp.Data

// ⚠️ 循环内的 err 遮蔽了外层的 err
for next, err := resp.Next(); next != nil && err == nil; {
    data = append(data, next.Data...)
}
if err != nil {  // 检查的是外层 err，不是循环终止原因
    return resp, err
}
return resp, err  // 返回初始 err，永远不是真正的错误
修复：外层用 `var err error`，循环用 `err = resp.Next()` 赋值而非声明

    connect(): Promise;
    send(msg: string): Promise;
}
const tuiClient = this.client as TuiClient;

function isTuiClient(x: unknown): x is TuiClient {

🟡 Rust Async TOCTOU 竞态

严重度：MEDIUM · 来源：MarkAI Code

// ❌ Time-of-Check-Time-of-Use
async fn verify_and_use_file(path: &Path) {
    let metadata = fs::metadata(path).await?; // Check
    if metadata.permissions().readonly() {
        return Err(Error::PermissionDenied);
    }
    // ⚠️ 时间窗口：权限可能在此被修改
    let content = fs::read_to_string(path).await?; // Use
}
// ✅ 修复：使用文件描述符
async fn verify_and_use_file_fd(fd: &File) {
    // 文件描述符在打开时锁定权限
    let content = fd.read_to_string().await?;
}

✅ Go 统一错误响应模式

// ✅ 领域错误型
type AppError struct {
    Code    int    // HTTP 状态码
    Message string // 用户可见消息
    Err     error  // 内部错误
}

func (e *AppError) Unwrap() error { return e.Err }

// ✅ 调用方使用 errors.As
var appErr *AppError
if errors.As(err, &appErr) {
    http.Error(w, appErr.Message, appErr.Code)
}

// ✅ 工厂函数
func NewNotFound(resource string) *AppError {
    return &AppError{Code: 404, Message: resource + " not found"}
}

🔴 Rust 认证竞态修复

来源：MarkAI Code Case Study

// ❌ 漏洞代码
async fn authenticate(creds: Credentials) {
    let user = db::find_user(&creds.username).await?;
    // ⚠️ 多线程可同时触发，生成多个 token
    if !verify_password(&creds.password, &user.hash).await? {
        return Err(InvalidCredentials);
    }
    let token = generate_token(&user).await?;
    db::update_last_login(&user.id).await?;
}
// ✅ 修复：使用事务
async fn authenticate(creds: Credentials) {
    let mut tx = db::begin_transaction().await?;
    let user = db::find_user_for_update(&mut tx, &creds.username).await?;
    if !verify_password(&creds.password, &user.hash).await? {
        tx.rollback().await?;
        return Err(InvalidCredentials);
    }
    let token = generate_token(&user).await?;
    db::update_last_login(&mut tx, &user.id).await?;
    tx.commit().await?;
}
